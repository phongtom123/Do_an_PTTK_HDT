CREATE DATABASE bleu CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE bleu;

-- Bảng USER
CREATE TABLE USER (
    user_ID INT AUTO_INCREMENT PRIMARY KEY,
    createDate DATE,
    name VARCHAR(255),
    email VARCHAR(255),
    account VARCHAR(255),
    password VARCHAR(255),
    level INT,
    progress VARCHAR(255)
);

-- Bảng Unit
CREATE TABLE Unit (
    unit_ID INT AUTO_INCREMENT PRIMARY KEY,
    unitName VARCHAR(255)
);

-- Bảng Learn (quan hệ N-N giữa USER và Unit)
CREATE TABLE Learn (
    learn_ID INT AUTO_INCREMENT PRIMARY KEY,  -- 🔹 Thêm ID riêng để dễ quản lý
    unit_ID INT NOT NULL,
    user_ID INT NOT NULL,
    UNIQUE (unit_ID, user_ID),                -- 🔹 Vẫn đảm bảo không trùng cặp
    FOREIGN KEY (unit_ID) REFERENCES Unit(unit_ID) ON DELETE CASCADE,
    FOREIGN KEY (user_ID) REFERENCES USER(user_ID) ON DELETE CASCADE
);

-- Bảng Vocab (từ vựng)
CREATE TABLE Vocab (
    word_ID INT AUTO_INCREMENT PRIMARY KEY,
    word VARCHAR(255),
    meaning VARCHAR(255),
    difficulty VARCHAR(255)
);

-- Bảng FlashCard (liên kết với từ vựng)
CREATE TABLE FlashCard (
    flashcard_ID INT AUTO_INCREMENT PRIMARY KEY, -- 🔹 ID riêng cho dễ thao tác
    word_ID INT NOT NULL,
    user_ID INT NOT NULL,
    UNIQUE (word_ID, user_ID),                   -- 🔹 Tránh trùng cặp người–từ
    FOREIGN KEY (word_ID) REFERENCES Vocab(word_ID) ON DELETE CASCADE,
    FOREIGN KEY (user_ID) REFERENCES USER(user_ID) ON DELETE CASCADE
);

-- Bảng Reading (bài đọc)
CREATE TABLE Reading (
    read_ID INT AUTO_INCREMENT PRIMARY KEY,
    readContent TEXT,
    unit_ID INT,
    FOREIGN KEY (unit_ID) REFERENCES Unit(unit_ID) ON DELETE CASCADE
);

-- Bảng Listening (bài nghe)
CREATE TABLE Listening (
    listen_ID INT AUTO_INCREMENT PRIMARY KEY,
    linkAudio TEXT,
    unit_ID INT,
    FOREIGN KEY (unit_ID) REFERENCES Unit(unit_ID) ON DELETE CASCADE
);

-- Bảng Question (câu hỏi)
CREATE TABLE Question (
    question_ID INT AUTO_INCREMENT PRIMARY KEY,
    content TEXT,
    unit_ID INT,
    explanation TEXT,
    result TEXT,
    FOREIGN KEY (unit_ID) REFERENCES Unit(unit_ID) ON DELETE CASCADE
);

-- Bảng UserAnswer (câu trả lời của người dùng)
CREATE TABLE UserAnswer (
    question_ID INT,
    user_ID INT,
    content TEXT,
    unit_ID INT,
    result BOOLEAN,
    PRIMARY KEY (question_ID, user_ID),
    FOREIGN KEY (question_ID) REFERENCES Question(question_ID) ON DELETE CASCADE,
    FOREIGN KEY (user_ID) REFERENCES USER(user_ID) ON DELETE CASCADE,
    FOREIGN KEY (unit_ID) REFERENCES Unit(unit_ID) ON DELETE CASCADE
);

-- ============================================
-- TẠO CƠ SỞ DỮ LIỆU
-- ============================================
CREATE DATABASE IF NOT EXISTS bleu
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE bleu;

-- ============================================
-- BẢNG Roles (Phân quyền)
-- ============================================
CREATE TABLE Roles (
  role_id INT AUTO_INCREMENT PRIMARY KEY,
  role_name VARCHAR(255) NOT NULL
);

-- ============================================
-- BẢNG Users (Người dùng)
-- ============================================
CREATE TABLE Users (
  user_id INT AUTO_INCREMENT PRIMARY KEY,
  user_name VARCHAR(50) NOT NULL UNIQUE,
  user_password VARCHAR(255) NOT NULL,
  user_role_id INT,
  user_rank INT DEFAULT 0,
  user_level INT DEFAULT 1,
  user_status TINYINT DEFAULT 1,
  FOREIGN KEY (user_role_id) REFERENCES Roles(role_id)
);

-- ============================================
-- BẢNG Units (Bài học lớn)
-- ============================================
CREATE TABLE Units (
  unit_id INT AUTO_INCREMENT PRIMARY KEY,
  unit_name VARCHAR(100) NOT NULL
);

-- ============================================
-- BẢNG Lessons (Bài học nhỏ)
-- Mỗi Lesson thuộc về một Unit
-- ============================================
CREATE TABLE Lessons (
  lesson_id INT AUTO_INCREMENT PRIMARY KEY,
  lesson_unit_id INT NOT NULL,
  lesson_name VARCHAR(100),
  FOREIGN KEY (lesson_unit_id) REFERENCES Units(unit_id)
);

-- ============================================
-- BẢNG Questions (Câu hỏi)
-- Mỗi Question thuộc về 1 Lesson
-- ============================================
CREATE TABLE Questions (
  question_id INT AUTO_INCREMENT PRIMARY KEY,
  question_lesson_id INT,
  question_unit_id INT,
  question_content VARCHAR(500),
  question_answer VARCHAR(100),
  question_type VARCHAR(50),
  FOREIGN KEY (question_lesson_id) REFERENCES Lessons(lesson_id),
  FOREIGN KEY (question_unit_id) REFERENCES Units(unit_id)
);

-- ============================================
-- BẢNG Question_options (Tùy chọn của câu hỏi)
-- ============================================
CREATE TABLE Question_options (
  question_option_id INT AUTO_INCREMENT PRIMARY KEY,
  question_option_question_id INT NOT NULL,
  option_1 VARCHAR(500),
  option_2 VARCHAR(500),
  option_3 VARCHAR(500),
  option_4 VARCHAR(500),
  FOREIGN KEY (question_option_question_id) REFERENCES Questions(question_id)
);

-- ============================================
-- BẢNG Listenings (Câu hỏi nghe)
-- ============================================
CREATE TABLE Listenings (
  listening_question_id INT PRIMARY KEY,
  listening_content VARCHAR(500),
  listening_audio VARCHAR(500),
  FOREIGN KEY (listening_question_id) REFERENCES Questions(question_id)
);

-- ============================================
-- BẢNG Readings (Câu hỏi đọc hiểu)
-- ============================================
CREATE TABLE Readings (
  reading_question_id INT PRIMARY KEY,
  reading_content VARCHAR(1000),
  FOREIGN KEY (reading_question_id) REFERENCES Questions(question_id)
);

-- ============================================
-- BẢNG Words (Flashcards)
-- Mỗi từ có thể thuộc về 1 Lesson
-- ============================================
CREATE TABLE Words (
  word_id INT AUTO_INCREMENT PRIMARY KEY,
  word VARCHAR(255) NOT NULL,
  word_meaning VARCHAR(255),
  word_status VARCHAR(30),
  word_difficulty VARCHAR(30),
  word_lesson_id INT,
  FOREIGN KEY (word_lesson_id) REFERENCES Lessons(lesson_id)
);

-- ============================================
-- BẢNG Learnings (Lưu tiến trình học theo Unit)
-- ============================================
CREATE TABLE Learnings (
  learning_id INT AUTO_INCREMENT PRIMARY KEY,
  learning_user_id INT NOT NULL,
  learning_unit_id INT NOT NULL,
  learning_finished_time TIMESTAMP NULL DEFAULT NULL,
  learning_score FLOAT DEFAULT 0,
  learning_is_pass BOOLEAN DEFAULT FALSE,
  learning_user_progress VARCHAR(50),
  learning_date DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (learning_user_id) REFERENCES Users(user_id),
  FOREIGN KEY (learning_unit_id) REFERENCES Units(unit_id)
);

-- ============================================
-- BẢNG Games (Kết quả game học tập)
-- ============================================
CREATE TABLE Games (
  game_id INT AUTO_INCREMENT PRIMARY KEY,
  game_user_id INT NOT NULL,
  correct_word_quantity INT DEFAULT 0,
  FOREIGN KEY (game_user_id) REFERENCES Users(user_id)
);

-- ============================================
-- BẢNG Answers (Câu trả lời của người học)
-- ============================================
CREATE TABLE Answers (
  answer_id INT AUTO_INCREMENT PRIMARY KEY,
  answer_question_id INT NOT NULL,
  answer_user_id INT NOT NULL,
  answer_user_answer VARCHAR(255),
  FOREIGN KEY (answer_question_id) REFERENCES Questions(question_id),
  FOREIGN KEY (answer_user_id) REFERENCES Users(user_id)
);







CREATE TABLE USER (
    user_ID INT AUTO_INCREMENT PRIMARY KEY,
    createDate DATE,
    name VARCHAR(255),
    email VARCHAR(255),
    account VARCHAR(255),
    password VARCHAR(255),
    level INT,
    progress VARCHAR(255),
    role BOOLEAN DEFAULT FALSE
);


CREATE TABLE Unit (
    unit_ID INT AUTO_INCREMENT PRIMARY KEY,
    unitName VARCHAR(255)
);


CREATE TABLE Learn (
    learn_ID INT AUTO_INCREMENT PRIMARY KEY,
    unit_ID INT NOT NULL,
    user_ID INT NOT NULL,
    is_pass BOOLEAN DEFAULT FALSE,
    finishTime INT,
    UNIQUE (unit_ID, user_ID),
    FOREIGN KEY (unit_ID) REFERENCES Unit(unit_ID) ON DELETE CASCADE,
    FOREIGN KEY (user_ID) REFERENCES USER(user_ID) ON DELETE CASCADE
);

CREATE TABLE Vocab (
    word_ID INT AUTO_INCREMENT PRIMARY KEY,
    word VARCHAR(255),
    meaning VARCHAR(255),
    difficulty VARCHAR(255)
);


CREATE TABLE FlashCard (
    flashcard_ID INT AUTO_INCREMENT PRIMARY KEY,
    word_ID INT NOT NULL,
    user_ID INT NOT NULL,
    UNIQUE (word_ID, user_ID),
    FOREIGN KEY (word_ID) REFERENCES Vocab(word_ID) ON DELETE CASCADE,
    FOREIGN KEY (user_ID) REFERENCES USER(user_ID) ON DELETE CASCADE
);

CREATE TABLE UserFC (
    flashcard_ID INT NOT NULL,
    user_ID INT NOT NULL,
    PRIMARY KEY (flashcard_ID, user_ID),
    FOREIGN KEY (flashcard_ID) REFERENCES FlashCard(flashcard_ID) ON DELETE CASCADE,
    FOREIGN KEY (user_ID) REFERENCES USER(user_ID) ON DELETE CASCADE
);

CREATE TABLE FCVocab (
    flashcard_ID INT NOT NULL,
    word_ID INT NOT NULL,
    synonym VARCHAR(255),
    PRIMARY KEY (flashcard_ID, word_ID),
    FOREIGN KEY (flashcard_ID) REFERENCES FlashCard(flashcard_ID) ON DELETE CASCADE,
    FOREIGN KEY (word_ID) REFERENCES Vocab(word_ID) ON DELETE CASCADE
);

CREATE TABLE Lesson (
    lesson_ID INT AUTO_INCREMENT PRIMARY KEY,
    unit_ID INT NOT NULL,
    lessonName VARCHAR(255),
    FOREIGN KEY (unit_ID) REFERENCES Unit(unit_ID) ON DELETE CASCADE
);

CREATE TABLE Reading (
    read_ID INT AUTO_INCREMENT PRIMARY KEY,
    lesson_ID INT NOT NULL,
    readContent TEXT,
    question_ID INT,
    FOREIGN KEY (lesson_ID) REFERENCES Lesson(lesson_ID) ON DELETE CASCADE
);

CREATE TABLE Listening (
    listen_ID INT AUTO_INCREMENT PRIMARY KEY,
    lesson_ID INT NOT NULL,
    linkAudio TEXT,
    question_ID INT,
    FOREIGN KEY (lesson_ID) REFERENCES Lesson(lesson_ID) ON DELETE CASCADE
);

CREATE TABLE Question (
    question_ID INT AUTO_INCREMENT PRIMARY KEY,
    content TEXT,
    unit_ID INT,
    explanation TEXT,
    result TEXT,
    FOREIGN KEY (unit_ID) REFERENCES Unit(unit_ID) ON DELETE CASCADE
);

CREATE TABLE UserAnswer (
    question_ID INT,
    user_ID INT,
    unit_ID INT,
    content TEXT,
    result BOOLEAN,
    PRIMARY KEY (question_ID, user_ID),
    FOREIGN KEY (question_ID) REFERENCES Question(question_ID) ON DELETE CASCADE,
    FOREIGN KEY (user_ID) REFERENCES USER(user_ID) ON DELETE CASCADE,
    FOREIGN KEY (unit_ID) REFERENCES Unit(unit_ID) ON DELETE CASCADE
);
